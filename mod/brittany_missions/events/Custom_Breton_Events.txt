    title = bri_missions.1.t
    desc = bri_missions.1.d
    picture = DIPLOMACY_eventPicture

    is_triggered_only = yes

    # Preview French Sphere
    option = {
        name = bri_missions.1.a
        custom_tooltip = bri_french_sphere_preview_tt

        clr_country_flag = bri_autonomous_path

        swap_non_generic_missions = yes

        hidden_effect = {
            if = {
                limit = {
                    ai = yes
                }
                clr_country_flag = bri_diplomacy_preview
            }
        }

        ai_chance = {
            factor = 1
        }
    }

    # Preview Autonomous Path
    option = {
        name = bri_missions.1.b
        custom_tooltip = bri_autonomous_preview_tt

        clr_country_flag = bri_french_sphere_path
        set_country_flag = bri_autonomous_path

        swap_non_generic_missions = yes

        hidden_effect = {
            if = {
                limit = {
                    ai = yes
                }
                clr_country_flag = bri_diplomacy_preview
            }
        }

        ai_chance = {
            factor = 1
        }
    }

    # Lock currently previewed branch
    option = {
        name = bri_missions.1.c

        trigger = {
            ai = no

            OR = {
                has_country_flag = bri_french_sphere_path
                has_country_flag = bri_autonomous_path
            }
        }

        custom_tooltip = bri_lock_diplomatic_path_tt

        clr_country_flag = bri_diplomacy_preview

        ai_chance = {
            factor = 0
        }
    }
}
